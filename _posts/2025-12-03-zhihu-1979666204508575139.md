    """
    Reward function that checks if the label ("Rise" or "Decrease") is present in the &lt;answer&gt; section.
    """
    rewards = []
    
    for completion, sol in zip(completions, solution):
        # 提取生成答案的内容
        content = completion[0]["content"]
        
        # 检查是否包含 &lt;answer&gt; 标签
        if "&lt;answer&gt;" in content and "&lt;/answer&gt;" in content:
            # 提取 &lt;answer&gt; 部分的内容
            start_idx = content.find("&lt;answer&gt;") + len("&lt;answer&gt;")
            end_idx = content.find("&lt;/answer&gt;")
            answer_section = content[start_idx:end_idx].strip()
            
            # 判断标准答案是否在 &lt;answer&gt; 部分中
            if sol in answer_section:
                reward = 1.0  # 如果标准答案在 &lt;answer&gt; 中，奖励为 1
            else:
                reward = 0.0  # 否则奖励为 0
        else:
            # 如果没有 &lt;answer&gt; 标签，直接给 0 奖励
            reward = 0.0
        
        rewards.append(reward)
    
    return rewards

ef get_cosine_scaled_reward_finance(
    min_value_wrong: float = -1.0,
    max_value_wrong: float = -0.5,
    min_value_correct: float = 0.5,
    max_value_correct: float = 1.0,
    max_len: int = 1000,
):
    def cosine_scaled_reward(completions, solution, **kwargs):
        """Reward function that assigns fixed rewards based on correctness."""
        contents = [completion[0]["content"] for completion in completions]
        rewards = []
        for content, sol in zip(contents, solution):
            # 检查 &lt;answer&gt; 部分是否包含标准答案
            if "&lt;answer&gt;" in content and "&lt;/answer&gt;" in content:
                start_idx = content.find("&lt;answer&gt;") + len("&lt;answer&gt;")
                end_idx = content.find("&lt;/answer&gt;")
                answer_section = content[start_idx:end_idx].strip()
                
                # 判断是否匹配标准答案
                is_correct = sol in answer_section
            else:
                is_correct = False
            
            # 根据正确性分配奖励值
            if is_correct:
                reward = max_value_correct  # 正确答案的最大奖励
            else:
                reward = min_value_wrong  # 错误答案的最小惩罚
            rewards.append(float(reward))
        return rewards
    return cosine_scaled_reward

def get_repetition_penalty_reward_finance(ngram_size: int, max_penalty: float):
    """
    Simplified repetition penalty function for stock prediction tasks.
    Since the generated answers are short labels ("Rise" or "Decrease"), 
    no actual repetition penalty is needed.
    """
    if max_penalty &gt; 0:
        raise ValueError(f"max_penalty {max_penalty} should not be positive")

        """
        Reward function that always returns 0 for stock prediction tasks.
        Args:
            completions: List of model completions
        """
        rewards = []
        for completion in completions:
            # 对于股票新闻任务，直接返回 0（无惩罚）
            rewards.append(0.0)
        return rewards
