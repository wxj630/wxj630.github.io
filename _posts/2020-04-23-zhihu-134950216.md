infinity = float('inf')


model = AbstractModel() # 这是一个抽象的Pyomo模型，因此需要提供set和参数值来初始化模型，见下面的“问题数据：通过dat文件读取”。


# Foods
model.F = Set()
# Nutrients
model.N = Set()


# Cost of each food
model.c    = Param(model.F, within=PositiveReals) # ci
# Amount of nutrient in each food
model.a    = Param(model.F, model.N, within=NonNegativeReals) # aij
# Lower and upper bound on each nutrient
model.Nmin = Param(model.N, within=NonNegativeReals, default=0.0) # Nmin-j
model.Nmax = Param(model.N, within=NonNegativeReals, default=infinity) # Nmax-j
# Volume per serving of food
model.V    = Param(model.F, within=PositiveReals) # Vi
# Maximum volume of food consumed
model.Vmax = Param(within=PositiveReals) # Vmax</code></pre></div><p data-pid="d8h2VA10">这些参数声明中使用了within选项来定义参数的预期属性。此信息用于对用于初始化参数组件的数据执行错误检查。</p><h2>决策变量</h2><p data-pid="GQ2RwUGF">Var组件用于定义决策变量:</p><div class="highlight"><pre><code class="language-text"># Number of servings consumed of each food
model.x = Var(model.F, within=NonNegativeIntegers) # 非负整数
 
目标函数
# Minimize the cost of food that is consumed
def cost_rule(model):
 return sum(model.c[i]*model.x[i] for i in model.F)
model.cost = Objective(rule=cost_rule)


约束条件
# Limit nutrient consumption for each nutrient
def nutrient_rule(model, j):
    value = sum(model.a[i,j]*model.x[i] for i in model.F)
 return model.Nmin[j] &lt;= value &lt;= model.Nmax[j]
model.nutrient_limit = Constraint(model.N, rule=nutrient_rule)


# Limit the volume of food consumed
def volume_rule(model):
 return sum(model.V[i]*model.x[i] for i in model.F) &lt;= model.Vmax
